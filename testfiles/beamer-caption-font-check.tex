\documentclass[notheorems]{beamer}
\usepackage{figmint}

\makeatletter
\newcommand*{\figmintprintfontsize}[2]{%
  \begingroup
    #2%
    \typeout{FIGMINT_FONT name=#1;size=\f@size}%
  \endgroup
}
\newcommand*{\figmintprintcurrentfontsize}[1]{%
  \typeout{FIGMINT_FONT name=#1;size=\f@size}%
}
\makeatother

\begin{document}

\figmintprintfontsize{caption}{\usebeamerfont{caption}}
\figmintprintfontsize{primary}{\figmintprimaryfont}
\figmintprintfontsize{secondary}{\figmintsecondaryfont}

\begin{frame}
  Beamer caption font check.

  \figmintset{figure width=2in, figure height=1.2in}
  \begin{figmintfigure}
    \begin{axis}[
      xmin=-0.5, xmax=0.5,
      ymin=0, ymax=1,
      xlabel={\figmintprintcurrentfontsize{axis-label}$x$},
      xtick={0},
      xticklabel={\figmintprintcurrentfontsize{tick}\pgfmathprintnumber{\tick}},
      nodes near coords={\figmintprintcurrentfontsize{node}0.5},
    ]
      \addplot+[figmint bar] coordinates {(0,0.5)};
      \addlegendentry{\figmintprintcurrentfontsize{legend}value}
    \end{axis}
  \end{figmintfigure}

  \figmintset{figure width=2in, figure height=1.2in}
  \begin{figmintfigure}
    \begin{axis}[
      figmint heatmap,
      figmint colorbar,
      xmin=-0.5, xmax=1.5,
      ymin=-0.5, ymax=1.5,
      colorbar style={
        ytick={0},
        yticklabel={\figmintprintcurrentfontsize{colorbar}\pgfmathprintnumber{\tick}},
      },
    ]
      \addplot+[matrix plot*,mesh/rows=2,mesh/cols=2,point meta=explicit] table[row sep=crcr,meta=z] {
        x y z\\
        0 0 0\\
        1 0 1\\
        0 1 0.5\\
        1 1 0.75\\
      };
    \end{axis}
  \end{figmintfigure}
\end{frame}

\end{document}
