\documentclass{article}
\usepackage[letterpaper,margin=0.55in]{geometry}
\usepackage{figmint}

\makeatletter
\newcommand*{\figmintprintgridstate}[1]{%
  \ifpgfplots@xmajorgrids
    \def\figmintxmajorgrids{true}%
  \else
    \def\figmintxmajorgrids{false}%
  \fi
  \ifpgfplots@ymajorgrids
    \def\figmintymajorgrids{true}%
  \else
    \def\figmintymajorgrids{false}%
  \fi
  \typeout{FIGMINT_GRID name=#1;x=\figmintxmajorgrids;y=\figmintymajorgrids}%
}
\newcommand*{\figmintprintcolormap}[1]{%
  \pgfkeysgetvalue{/pgfplots/colormap name}\figmintcolormapname
  \typeout{FIGMINT_COLORMAP name=#1;value=\figmintcolormapname}%
}
\makeatother
