\documentclass[10pt]{article}
\usepackage{fontspec}
\setmainfont{TeX Gyre Termes}
\usepackage[letterpaper,margin=0.75in]{geometry}
\usepackage{figmint}

% The legend fixtures in tests/fixtures/legend are shared with the matplotlib
% backend; tests/fixtures/legend/cases.tsv holds the expected placements.
\pgfplotsset{
    table/col sep=tab,
    fixture/.style={
        figmint,
        figmint legend report=#1,
        scale only axis,
        width=8cm,
        height=5cm,
        xmin=0,
        xmax=1,
        ymin=0,
        ymax=1,
        xtick={0,0.5,1},
        ytick={0,0.5,1},
    },
}
\newcommand{\fixturepolygon}[2]{%
    \addplot+[fill, draw=none, area legend] table {tests/fixtures/legend/#1/#2.tsv} -- cycle;
    \addlegendentry{#2}%
}
\newcommand{\fixturebars}[2]{%
    \addplot+[ybar, bar width=0.08, fill, draw=none, area legend] table {tests/fixtures/legend/#1/#2.tsv};
    \addlegendentry{#2}%
}
\newcommand{\fixtureline}[2]{%
    \addplot+[figmint line] table {tests/fixtures/legend/#1/#2.tsv};
    \addlegendentry{#2}%
}

% An axis without a legend keeps the per-point processing of PGFPlots
% untouched, which keeps its Lua backend available.
\begin{tikzpicture}
    \begin{axis}[fixture=no-legend]
        \addplot+[figmint line] coordinates {(0,0.2) (1,0.8)};
        \pgfplotsextra{%
            \pgfkeysgetvalue{/pgfplots/execute for finished point}\figmintcheckhook
            \typeout{FIGMINT_HOOK name=no-legend;finished-point=[\figmintcheckhook]}%
        }
    \end{axis}
\end{tikzpicture}
