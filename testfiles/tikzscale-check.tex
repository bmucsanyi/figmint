\newsavebox{\figminttikzscalebox}
\newlength{\figminttikzscaledelta}
\newcommand*{\figmintmeasuretikzscale}[3]{%
  \begin{figure}[!htbp]
    \centering
    \sbox{\figminttikzscalebox}{\includegraphics[width=\linewidth,axisratio=#2]{#1}}%
    \setlength{\figminttikzscaledelta}{\dimexpr\wd\figminttikzscalebox-\linewidth\relax}%
    \ifdim\figminttikzscaledelta<0pt
      \setlength{\figminttikzscaledelta}{-\figminttikzscaledelta}%
    \fi
    \ifdim\figminttikzscaledelta>\hfuzz
      \PackageError{figmint-test}{#3 width differs from linewidth}{Tikzscale must fit #3 to linewidth.}%
    \fi
    \typeout{FIGMINT_TIKZSCALE_WIDTH name=#3 actual=\the\wd\figminttikzscalebox target=\the\linewidth}%
    \usebox{\figminttikzscalebox}
    \caption{Tikzscale check for #3.}
  \end{figure}
}
